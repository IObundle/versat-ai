module Top_Add #(X=1) (){
   VRead inputs_0[X];
   VRead inputs_1[X];
   VWrite output[X];
   F_Add adder[X];
#
   for i 0..X {
      {inputs_0[i],inputs_1[i]} -> adder[i]:0..1;
      adder[i] -> output[i];
   }
##
   config Linear(addr: Buffer,size: Dyn){
      gen i 0..X{
         for x RangeLow(0,size,X,i)..RangeHigh(0,size,X,i) {
            inputs_0[i] = addr[x];
         }
      }
   }

   config Broadcast(addr: Buffer,size: Dyn,enable){
      gen i 0..X{
         for x RangeLow(0,size,X,i)..RangeHigh(0,size,X,i) {
            inputs_1[i] = addr[x * enable];
         }
      }
   }

   config Output(addr: Buffer,size){
      gen i 0..X{
         for x RangeLow(0,size,X,i)..RangeHigh(0,size,X,i) {
            output[i] = addr[x];
         }
      }
   }

   state NumberUnits{
      val = X;
   }
}

##

      }
   }
}

module Top_Maxpool(){
   VRead features;
   F_AccumMax accum;
   VWrite output;
#   
   features -> accum;
   accum -> output;
##
   config Features(addr: Buffer,startX,startY,startOffset,kernelW,kernelH,imageWidth,sizeW,sizeH,strideW,strideH){
      for y  0..sizeH {
         for x  0..sizeW {
            for ky 0..kernelH {
               for kx 0..kernelW {
                  features = addr[startOffset + (y * strideH + startY + ky) * imageWidth + x * strideW + startX + kx];
               }
            }
         }
      }
   }

   config Output(addr: Buffer,startX,startY,startOffset,sizeW,sizeH,outputImageW,stride){
      for y 0..sizeH {
         for x 0..sizeW {
            output = addr[Duty(((y + startY) * outputImageW + (x + startX) + startOffset),stride)];
         }
      }
   }
}

module Top_AveragePool(){
   VRead features;
   MyFloatAccum averagePool_accum;
   F_Mul inversedDiv; // Easier to multiply by (1/divisor) than to bring a div unit.
   Const invertedDivisor;
   VWrite output;
#   
   features -> averagePool_accum;
   {averagePool_accum,invertedDivisor} -> inversedDiv:0..1;
   inversedDiv -> output;
##
   config Features(addr: Buffer,startX,startY,startOffset,kernelW,kernelH,imageWidth,sizeW,sizeH,strideW,strideH){
      for y  0..sizeH {
         for x  0..sizeW {
            for ky 0..kernelH {
               for kx 0..kernelW {
                  features = addr[startOffset + (y * strideH + startY + ky) * imageWidth + x * strideW + startX + kx];
               }
            }
         }
      }
   }

   config Output(addr: Buffer,startX,startY,startOffset,sizeW,sizeH,outputImageW,stride){
      for y 0..sizeH {
         for x 0..sizeW {
            output = addr[Duty(((y + startY) * outputImageW + (x + startX) + startOffset),stride)];
         }
      }
   }
}

module Top_Conv #(X = 1,Y = 1) (){
   VRead #(.ADDR_W(17)) featuresConv[Y][X];
   VRead #(.ADDR_W(17)) weightsConv[Y][X];
   VRead bias;

   F_Mul muller[Y][X];
   F_Add addBias;
   MyFloatAccum myAccum;
   VWrite outputConv;
#   
   for y 0..Y {
      for x 0..X {
         {featuresConv[y][x],weightsConv[y][x]} -> muller[y][x]:0..1;
      }
   }
   reduced = Reduce(muller[0..Y][0..X],F_Add);
   reduced -> myAccum;

   {myAccum,bias} -> addBias:0..1;
   addBias -> outputConv;
##
   // Input image W is only used to calculate the position 

   config FeaturesWeightsOutputs(inputX : Buffer,inputW : Buffer,outAddr : Buffer,
      actualKernelW : Fixed,actualKernelH : Fixed,sizeC : Fixed,
      outputH : Dyn,outputW : Dyn,featuresAmount : Dyn,

      startX,startY,
      kernelStartX,kernelStartY,
      outCStart, // D is outC
      outStartX,outStartY,
      inputImageW,inputImagePerGroup,
      startC,
      kernelW,kernelH,
      outputWidth,stride,outputChannels, strideW, strideH){

      myAccum.strideMinusOne = RangeSize(0,actualKernelH,Y,0) * RangeSize(0,actualKernelW,X,0) * inputImagePerGroup - 1;

      gen y 0..Y {
         gen x 0..X {
            for outerY 0..outputH {
               for outerX 0..outputW {
                  for outC 0..featuresAmount {
                     for ky RangeLow(0,actualKernelH,Y,y)..RangeHigh(0,actualKernelH,Y,y) {
                        for kx RangeLow(0,actualKernelW,X,x)..RangeHigh(0,actualKernelW,X,x) {
                           for inC 0..sizeC {
                              featuresConv[y][x] = inputX[(startY + ky + outerY * strideH) * inputImageW * inputImagePerGroup + (startX + kx + outerX * strideW) * inputImagePerGroup + (inC + startC)];
                              featuresConv[y][x].workSize = (RangeSize(0,actualKernelH,Y,0) * RangeSize(0,actualKernelW,X,0));
                              featuresConv[y][x].work = (RangeSize(0,actualKernelH,Y,y) * RangeSize(0,actualKernelW,X,x));

                              weightsConv[y][x] = inputW[(outC + outCStart) * inputImagePerGroup * kernelW * kernelH + (inC + startC) * kernelW * kernelH + (ky + kernelStartY) * kernelW + (kx + kernelStartX)];
                              weightsConv[y][x].workSize = (RangeSize(0,actualKernelH,Y,0) * RangeSize(0,actualKernelW,X,0));
                              weightsConv[y][x].work = (RangeSize(0,actualKernelH,Y,y) * RangeSize(0,actualKernelW,X,x));
                           }
                        }
                     }
                  }
               }
            }
         }
      }

      for y 0..outputH {
         for x 0..outputW {
            for inC 0..featuresAmount {
               outputConv = outAddr[Duty(((y + outStartY) * outputWidth * outputChannels + (x + outStartX) * outputChannels + outCStart + inC),
               RangeSize(0,actualKernelH,Y,0) * RangeSize(0,actualKernelW,X,0) * inputImagePerGroup)];
            }
         }
      }
   }

   config Bias(addr : Buffer,size,stride,outerXSize,outerYSize,actualKernelH,actualKernelW,inputImagePerGroup){
      for outerY 0..outerYSize {
         for outerX 0..outerXSize {
            for x 0..size {
               bias = addr[Duty(x,RangeSize(0,actualKernelH,Y,0) * RangeSize(0,actualKernelW,X,0) * inputImagePerGroup)];
            }
         }
      }
   }

   state NumberUnits{
      gridX = X;
      gridY = Y;
   }
}

module Top_MatMul(){
   VRead leftRow;
   VRead rightRow;
   F_Mul mat_muller;
   MyFloatAccum myAccum;
   VWrite output;
#
   {leftRow,rightRow} -> mat_muller:0..1;
   mat_muller -> myAccum;
   myAccum -> output;
##
   config Simple(leftAddr: Buffer,rightAddr: Buffer,size: Fixed,linesToProcess: Dyn){
      for y 0..linesToProcess{
         for x 0..size {
            leftRow = leftAddr[x];
            rightRow = rightAddr[y * size + x];
         }
      }
   }   

   config Output(addr: Buffer,size,stride){
      for x 0..size {
         output = addr[Duty(x,stride)];
      }
   }
}

module Top_BatchNormalization(){
   VRead x;
   VWrite o;

   Const A;
   Const B;

   F_Mul mul;
   F_Add add;
#
   {x,A} -> mul:0..1;
   {mul,B} -> add:0..1;

   add -> o;
##
   config Simple(input : Buffer,output : Buffer,index,size : Dyn,aVal,bVal) {
      A.constant = aVal;
      B.constant = bVal;
      for i 0..size {
         x = input[index + i];
         o = output[index + i];
      }
   }
}

module Log(in){
   ExtractExpAsFloat expAsFloat;
   ExtractMantissa   mantissaIndex;

   LookupTable #(.ADDR_W(14)) mantissaLookup;

   Const log2Value;

   F_Mul mul;
   F_Add adder;
#
   in -> expAsFloat;
   in -> mantissaIndex;

   mantissaIndex -> mantissaLookup;

   {expAsFloat,log2Value} -> mul:0..1;
   {mul,mantissaLookup} -> adder:0..1;

   adder -> out;
##
   config LoadLog2(log2Val){
      log2Value.constant = log2Val;
   }

   mem LoadMantissa(addr: Buffer, width){
      for x 0..width {
         mantissaLookup[x] = addr[x];
      }
   } 
}

module Exp(in){
   ExtractRealPart ext;
   F_Neg neg;
   F_Add asFraction;
   F_IsNeg isNeg;

   FracIndex frac;

   // Index bit is the precision.
   AddSignToIndex #(.INDEX_BIT(12)) fracIndex;

   LookupTable #(.ADDR_W(10)) expLookup;
   LookupTable #(.ADDR_W(15)) fracLookup;

   F_Mul mul;
#
   in -> ext;
   in -> isNeg;

   ext:1 -> neg;

   {in,neg} -> asFraction:0..1;

   asFraction -> frac;

   {frac,isNeg} -> fracIndex:0..1;

   ext:0 -> expLookup;
   fracIndex -> fracLookup;

   {expLookup,fracLookup} -> mul:0..1;

   mul -> out;   
##
   mem LoadExp(addr: Buffer, width){
      for x 0..width {
         expLookup[x] = addr[x];
      }
   }   

   mem LoadFrac(addr: Buffer, width){
      for x 0..width {
         fracLookup[x] = addr[x];
      }
   }
}

}

module Pow(base,power){
   // base ^ power  = exp(power * ln(base))
   Exp exp;
   Log log;
   F_Mul mul;
#
   base -> log;
   {log,power} -> mul:0..1;

   mul -> exp;

   exp -> out;
##
   config Init(log2Val){
      log.LoadLog2(log2Val);
   }

   mem LoadMantissa(addr: Buffer, width){
      log.LoadMantissa(addr,width);
   }
   mem LoadExp(addr: Buffer,width){
      exp.LoadExp(addr,width);
   }
   mem LoadFrac(addr: Buffer,width){
      exp.LoadFrac(addr,width);
   }
}

module Top_Gemm(){
   VRead #(.ADDR_W(17)) gemmLeftRow; 
   VRead #(.ADDR_W(17)) gemmRightRow;
   PipelineRegister leftPipe;
   PipelineRegister rightPipe;
   F_Mul mat_muller;
   MyFloatAccum myAccum;
   VWrite #(.ADDR_W(17)) output;

   Const alpha;
   F_Mul alphaMul;

   Const cValue;
   F_Add adder;
#
   gemmLeftRow -> leftPipe;
   gemmRightRow -> rightPipe;

   {leftPipe,rightPipe} -> mat_muller:0..1;
   mat_muller -> myAccum;

   {myAccum,alpha} -> alphaMul:0..1;

   {alphaMul,cValue} -> adder:0..1;

   adder -> output;
##

   config CValue(val){

         gemmRightRow = rightAddr[x];
      }
   }

   config Output(addr: Buffer,size,stride){
      for x 0..size {
         output = addr[Duty(x,stride)];
      }
   }
}

module Top_LRN(){
   Const constK;
   Const constA;
   Const constB;

   MyFloatAccum accum;
   F_Mul pow2;
   F_Mul alphaMul;

   F_Add adder;

   Pow pow;

   VRead in;
   VWrite output;
#
   {in,in} -> pow2:0..1;
   pow2 -> accum;

   {accum,constA}    -> alphaMul:0..1;
   {alphaMul,constK} -> adder:0..1;

   {adder,constB} -> pow:0..1;

   pow -> output;
##
   config InitConsts(valA,valB,valK,log2Val){
      constA.constant = valA;
      constB.constant = valB;
      constK.constant = valK;
      pow.Init(log2Val);
   }

   config Simple(inAddr: Buffer,outAddr: Buffer,lowerBound,width){
      accum.strideMinusOne = width - 1;
      for x 0..width{
         in = inAddr[lowerBound + x];
      }
      for x 0..1 {
         output = outAddr[x];
      }
      output.extra_delay = width - 1;
   }

   mem LoadMantissa(addr: Buffer, width){
      pow.LoadMantissa(addr,width);
   }
   mem LoadExp(addr: Buffer,width){
      pow.LoadExp(addr,width);
   }
   mem LoadFrac(addr: Buffer,width){
      pow.LoadFrac(addr,width);
   }
}
